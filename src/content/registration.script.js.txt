
        const steps = document.querySelectorAll('.step-content');
        const nextBtn = document.getElementById('next-btn');
        const prevBtn = document.getElementById('prev-btn');
        const progressBar = document.getElementById('progress-bar');
        const progressPercentage = document.getElementById('progress-percentage');
        const stepCounter = document.getElementById('step-counter');
        const stepTitle = document.getElementById('step-title');
        const successScreen = document.getElementById('success-screen');
        const form = document.getElementById('reg-form');

        let currentStep = 1;
        const totalSteps = 9;

        const stepTitles = [
            "Personal Identity",
            "Guardian Details",
            "Academic Foundation",
            "Global Ambitions",
            "Standardized Metrics",
            "Financial Strategy",
            "Professional Context",
            "Service Selection",
            "Final Affirmation"
        ];

        function updateUI() {
            // Visibility
            steps.forEach(step => {
                const s = parseInt(step.dataset.step);
                if (s === currentStep) {
                    step.classList.remove('hidden');
                    step.classList.add('animate-in', 'fade-in', 'slide-in-from-right-4', 'duration-500');
                } else {
                    step.classList.add('hidden');
                }
            });

            // Progress Bar
            const percent = (currentStep / totalSteps) * 100;
            progressBar.style.width = `${percent}%`;
            progressPercentage.textContent = `${Math.round(percent)}% Complete`;
            stepCounter.textContent = `Step 0${currentStep} / 09`;
            stepTitle.textContent = stepTitles[currentStep - 1];

            // Nav Buttons
            if (currentStep === 1) {
                prevBtn.classList.add('opacity-0', 'pointer-events-none');
            } else {
                prevBtn.classList.remove('opacity-0', 'pointer-events-none');
            }

            if (currentStep === totalSteps) {
                nextBtn.textContent = "FINALIZE & SUBMIT";
            } else {
                nextBtn.textContent = "CONTINUE";
            }
        }

        nextBtn.addEventListener('click', () => {
            if (currentStep < totalSteps) {
                currentStep++;
                updateUI();
                window.scrollTo({ top: 0, behavior: 'smooth' });
            } else {
                // Handle Submission
                const declaration = document.getElementById('declare');
                if (!declaration.checked) {
                    declaration.parentElement.classList.add('animate-bounce');
                    setTimeout(() => declaration.parentElement.classList.remove('animate-bounce'), 1000);
                    return;
                }
                
                successScreen.classList.remove('hidden');
                successScreen.classList.add('animate-in', 'fade-in', 'duration-1000');
            }
        });

        prevBtn.addEventListener('click', () => {
            if (currentStep > 1) {
                currentStep--;
                updateUI();
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }
        });

        // Micro-interactions for floating labels
        document.querySelectorAll('.floating-label-input input, .floating-label-input select, .floating-label-input textarea').forEach(el => {
            el.addEventListener('focus', () => {
                el.parentElement.querySelector('label').style.color = '#f5bd5c';
            });
            el.addEventListener('blur', () => {
                if (!el.value) {
                    el.parentElement.querySelector('label').style.color = '';
                }
            });
        });
