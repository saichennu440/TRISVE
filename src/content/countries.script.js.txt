(function () {

    function initGlobalFootprint() {

        const section = document.getElementById("global-footprint");

        if (!section) {
            console.warn("Global Footprint section not found.");
            return;
        }

        const pins = section.querySelectorAll(".gf-pin");

        if (!pins.length) {
            console.warn("No Global Footprint pins found.");
            return;
        }

        console.log("Global Footprint initialized.");
        console.log("Pins found:", pins.length);


        /* ==========================================
           SCROLL TO COUNTRY CARD
        ========================================== */

        function scrollToCountry(key) {

            const countryCard = document.getElementById(
                "country-" + key
            );

            console.log("Selected country:", key);

            if (!countryCard) {
                console.warn(
                    "Country card not found:",
                    "country-" + key
                );
                return;
            }

            const navbarOffset = 100;

            const cardPosition =
                countryCard.getBoundingClientRect().top +
                window.scrollY -
                navbarOffset;

            window.scrollTo({
                top: cardPosition,
                behavior: "smooth"
            });


            /* Highlight the selected country card */

            countryCard.classList.add(
                "country-card-highlight"
            );

            setTimeout(function () {

                countryCard.classList.remove(
                    "country-card-highlight"
                );

            }, 1500);
        }


        /* ==========================================
           MAP PIN CLICK
        ========================================== */

        pins.forEach(function (pin) {

            pin.addEventListener("click", function (event) {

                event.preventDefault();
                event.stopPropagation();

                const key =
                    pin.getAttribute("data-location");

                if (!key) {
                    console.warn(
                        "Pin has no data-location:",
                        pin
                    );
                    return;
                }

                console.log("PIN CLICKED:", key);


                /* Active pin */

                pins.forEach(function (item) {

                    item.classList.remove("is-active");

                    item.setAttribute(
                        "aria-pressed",
                        "false"
                    );

                });

                pin.classList.add("is-active");

                pin.setAttribute(
                    "aria-pressed",
                    "true"
                );


                /* Scroll to matching country card */

                scrollToCountry(key);

            });

        });


        /* ==========================================
           OPTIONAL DESTINATION BUTTONS
        ========================================== */

        const destinationButtons =
            section.querySelectorAll(".gf-destination");

        destinationButtons.forEach(function (button) {

            button.addEventListener("click", function () {

                const key =
                    button.getAttribute("data-location");

                if (key) {
                    scrollToCountry(key);
                }

            });

        });


        /* ==========================================
           CARD HIGHLIGHT STYLE
        ========================================== */

        if (!document.getElementById("gf-card-highlight-style")) {

            const style =
                document.createElement("style");

            style.id =
                "gf-card-highlight-style";

            style.textContent = `
                .country-card-highlight {
                    outline: 2px solid rgba(245, 189, 92, 0.75);
                    outline-offset: -2px;
                    transition: outline 0.3s ease;
                }
            `;

            document.head.appendChild(style);
        }

    }


    /* ==========================================
       INITIALIZE

       HtmlPage executes this file after the page
       HTML has already been inserted.
    ========================================== */

    initGlobalFootprint();

})();

        /* =================================================
           UPDATE CONTENT
        ================================================= */

        function updatePreview(
            location
        ) {

            const country =
                countryData[
                    location
                ];


            if (
                !country
            ) {

                return;

            }


            previewImage.src =
                country.image;


            previewImage.alt =
                country.name;


            previewFlag.src =
                country.flag;


            previewFlag.alt =
                country.name +
                " flag";


            previewCountry.textContent =
                country.name;


            previewDescription.textContent =
                country.description;

        }


        /* =================================================
           POSITION CARD
        ================================================= */

        function positionPreview(
            pin
        ) {

            const canvasRect =
                mapCanvas
                    .getBoundingClientRect();


            const pinRect =
                pin
                    .getBoundingClientRect();


            const previewWidth =
                preview.offsetWidth ||
                300;


            const previewHeight =
                preview.offsetHeight ||
                270;


            const x =
                pinRect.left -
                canvasRect.left +
                pinRect.width / 2;


            const y =
                pinRect.top -
                canvasRect.top +
                pinRect.height / 2;


            const gap =
                22;


            /*
             Default:
             show card to right
            */

            let left =
                x +
                gap;


            let top =
                y -
                previewHeight / 2;


            /*
             For countries on right side
             show card on left
            */

            if (
                left +
                previewWidth >
                mapCanvas.clientWidth -
                20
            ) {

                left =
                    x -
                    previewWidth -
                    gap;

            }


            /*
             Prevent top overflow
            */

            if (
                top <
                20
            ) {

                top =
                    20;

            }


            /*
             Prevent bottom overflow
            */

            if (
                top +
                previewHeight >
                mapCanvas.clientHeight -
                20
            ) {

                top =
                    mapCanvas.clientHeight -
                    previewHeight -
                    20;

            }


            preview.style.left =
                left +
                "px";


            preview.style.top =
                top +
                "px";

        }


        /* =================================================
           SHOW
        ================================================= */

        function showPreview(
            pin
        ) {

            const location =
                pin.dataset.location;


            updatePreview(
                location
            );


            /*
             Make temporarily measurable
            */

            preview.style.display =
                "block";


            preview.style.visibility =
                "hidden";


            requestAnimationFrame(
                function () {

                    positionPreview(
                        pin
                    );


                    preview.style.visibility =
                        "";


                    preview.classList.add(
                        "is-visible"
                    );


                    preview.setAttribute(
                        "aria-hidden",
                        "false"
                    );

                }
            );

        }


        /* =================================================
           HIDE
        ================================================= */

        function hidePreview() {

            preview.classList.remove(
                "is-visible"
            );


            preview.setAttribute(
                "aria-hidden",
                "true"
            );

        }


        /* =================================================
           EVENTS
        ================================================= */

        pins.forEach(
            function (
                pin
            ) {


                pin.addEventListener(
                    "mouseenter",
                    function () {

                        showPreview(
                            pin
                        );

                    }
                );


                pin.addEventListener(
                    "mouseleave",
                    hidePreview
                );


                /*
                 Keyboard
                */

                pin.addEventListener(
                    "focus",
                    function () {

                        showPreview(
                            pin
                        );

                    }
                );


                pin.addEventListener(
                    "blur",
                    hidePreview
                );

            }
        );


        /*
         Recalculate position if
         window changes size.
        */

        window.addEventListener(
            "resize",
            function () {

                const activePin =
                    document.querySelector(
                        "#global-footprint .gf-pin:hover, #global-footprint .gf-pin:focus"
                    );


                if (
                    activePin
                ) {

                    positionPreview(
                        activePin
                    );

                }

            }
        );

    }
